#โครงสร้างไฟล์
  src/middleware.ts           กันเส้นทาง
  src/lib/session.ts          สร้าง/ตรวจ JWT (ใช้ได้ทั้ง Edge และ Node)
  src/lib/session-cookie.ts   อ่าน/เขียน/ลบ session cookie
  src/lib/auth.ts             เทียบรหัสผ่าน + rate limit
  src/app/actions.ts          Server Action: login / logout
  src/app/page.tsx            หน้าหลัก (ต้องล็อกอิน)
  src/app/login/              หน้า login + ฟอร์ม
